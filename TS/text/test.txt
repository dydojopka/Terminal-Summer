label prologue:
    #$ persistent.sprite_time = "day"
    scene bg int_bus with esflash
    show sl normal_pioneer_far size normal at center with dissolve
    window show
    "В глаза ударил яркий дневной свет."
    slp "что <i>бы</i> ты <b>сечас</b> выбрал?"
    #window hide
    menu
    {
        "sl"
        {
            #window show
            $lp_sl += 1
            $lp_un -= 2
            $lp_dv -= 3
            $lp_us -= 4
            "sl+=1 un-=2 dv-=3 us-=4"
        }
        "un"
        {
            $lp_sl -= 1
            $lp_un += 2
            $lp_dv -= 3
            $lp_us -= 4
            "sl -= 1 un += 2\ dv -= 3\n us -= 4"
        }
        "dv"
        {
            #window show
            $lp_sl -= 1
            $lp_un -= 2
            $lp_dv += 3
            $lp_us -= 4
            "sl-=1 un-=2 dv+=3 us-=4"
        }
        "us"
        {
            $lp_sl -= 1
            $lp_un -= 2
            $lp_dv -= 3
            $lp_us += 4
            "sl-=1 un-=2 dv-=3 us+=4"
        }
    }
    slp "Расчек флагов prologue"
    menu
    {
        "Да"
        {
            $prologue = 1
            "prologue = 1"
        }
        "Нет"
        {
            $prologue = 0
            "prologue = False"
        }
    }
    slp "Расчек флагов D1_KEYS"
        menu
        {
            "Да"
            {
                $d1_keys = true
                "D1_KEYS = 1"
            }
            "Нет"
            {
                $d1_keys = 0
                "D1_KEYS = 0"
            }
        }
    "Переход на день 1"
    pause 2
    load day1
